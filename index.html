/** * API สำหรับหน้า IT Asset ที่เปิดจาก GitHub Pages * วางไฟล์นี้ในโปรเจกต์ Apps Script ที่ผูกกับชีต Asset * * ตั้งค่าก่อนใช้: * 1) Project Settings -> Script Properties -> เพิ่ม TOKEN = รหัสผ่านที่ตั้งเอง (ยาว ๆ เดายาก) * 2) Deploy -> New deployment -> Web app * Execute as: Me * Who has access: Anyone * 3) คัดลอก Web app URL (ลงท้าย /exec) ไปใส่ที่ API_URL ใน index.html * * ถ้าโปรเจกต์มี doPost อยู่แล้ว ให้รวมโค้ดเข้าด้วยกัน (มี doPost ได้ตัวเดียว) * ถ้ามี doGet เดิมอยู่แล้ว ไม่ต้องแก้ คงไว้ได้เลย */ var SHEET_NAME = 'Asset01'; var HISTORY_NAME = 'History'; var ASSET_COL = 'IT Asset No'; var TZ = 'Asia/Bangkok'; // ชื่อฟังก์ชันที่หน้าเว็บเรียก -> ฟังก์ชันจริงในไฟล์นี้ var API = { getData: apiGetData, saveRow: apiSaveRow, getHistory: apiGetHistory }; // เปิด URL /exec ในเบราว์เซอร์ตรง ๆ (GET) จะมาที่ฟังก์ชันนี้ ใช้เช็กว่า API ทำงานอยู่ // ถ้าโปรเจกต์มี doGet เดิมอยู่แล้ว ให้ลบอันนี้ออก (มี doGet ได้ตัวเดียว) function doGet() { return ContentService.createTextOutput('IT Asset API is running') .setMimeType(ContentService.MimeType.TEXT); } function doPost(e) { var out; try { var req = JSON.parse(e.postData.contents); var token = PropertiesService.getScriptProperties().getProperty('TOKEN'); if (!token || req.key !== token) { out = { ok: false, error: 'unauthorized' }; } else if (!API.hasOwnProperty(req.fn)) { out = { ok: false, error: 'unknown function' }; } else { out = { ok: true, data: API[req.fn].apply(null, (req.args || []).concat([req.editor || ''])) }; } } catch (err) { out = { ok: false, error: String(err && err.message || err) }; } return ContentService.createTextOutput(JSON.stringify(out)) .setMimeType(ContentService.MimeType.JSON); } function cell_(v) { if (v instanceof Date) return Utilities.formatDate(v, TZ, 'dd/MM/yyyy'); return v; } function apiGetData() { var sh = SpreadsheetApp.getActive().getSheetByName(SHEET_NAME); var vals = sh.getDataRange().getValues(); var headers = vals[0].map(function (h) { return String(h).trim(); }); var a = headers.indexOf(ASSET_COL); var rows = []; for (var r = 1; r < vals.length; r++) { if (a >= 0 && String(vals[r][a]).trim() === '') continue; // ข้ามแถวว่าง rows.push({ n: r + 1, v: vals[r].map(cell_) }); // n = เลขแถวจริงในชีต } return { headers: headers, rows: rows }; } // changes = { "<ลำดับคอลัมน์ เริ่มที่ 0>": "ค่าใหม่" } function apiSaveRow(n, assetNo, changes, editor) { var lock = LockService.getScriptLock(); lock.waitLock(20000); try { var ss = SpreadsheetApp.getActive(); var sh = ss.getSheetByName(SHEET_NAME); var hist = ss.getSheetByName(HISTORY_NAME); var headers = sh.getRange(1, 1, 1, sh.getLastColumn()).getValues()[0] .map(function (h) { return String(h).trim(); }); var a = headers.indexOf(ASSET_COL); var row = sh.getRange(n, 1, 1, headers.length).getValues()[0]; // กันเขียนผิดแถว ถ้ามีคนเพิ่ม/ลบแถวระหว่างที่เปิดหน้าค้างไว้ if (String(row[a]).trim() !== String(assetNo).trim()) { throw new Error('ข้อมูลในชีตเปลี่ยนไปแล้ว กรุณารีเฟรชหน้าแล้วลองใหม่'); } var now = new Date(), log = [], saved = 0; Object.keys(changes).forEach(function (k) { var i = Number(k); if (!(i >= 0 && i < headers.length)) return; var oldV = row[i], newV = changes[k]; if (String(oldV) === String(newV)) return; sh.getRange(n, i + 1).setValue(newV); log.push([now, assetNo, headers[i], cell_(oldV), newV, editor || '']); saved++; }); if (log.length && hist) { hist.getRange(hist.getLastRow() + 1, 1, log.length, 6).setValues(log); } return { saved: saved }; } finally { lock.releaseLock(); } } function apiGetHistory(assetNo) { var hist = SpreadsheetApp.getActive().getSheetByName(HISTORY_NAME); if (!hist || hist.getLastRow() < 2) return []; var vals = hist.getRange(2, 1, hist.getLastRow() - 1, 6).getValues(); return vals .filter(function (r) { return String(r[1]).trim() === String(assetNo).trim(); }) .reverse() // ใหม่สุดขึ้นก่อน .map(function (r) { var d = r[0] instanceof Date ? Utilities.formatDate(r[0], TZ, 'dd/MM/yyyy HH:mm:ss') : r[0]; return [d, r[1], r[2], r[3], r[4], r[5]]; }); }
