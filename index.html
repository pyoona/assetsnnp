/** * IT Asset SNNP - Apps Script backend (Code.gs) * * รองรับการเรียก 2 แบบ ตามที่หน้า index.html ใช้: * 1) เปิดผ่าน Apps Script เอง -> google.script.run.gs_getData(key, who) ฯลฯ * 2) เปิดจากที่อื่น (GitHub Pages ฯลฯ) -> POST ไปที่ Web app URL (/exec) * body = {fn, args, key, editor} -> ตอบ {ok:true, data} หรือ {ok:false, error} * * ฟังก์ชันที่หน้าเว็บเรียก: getData, saveRow, getHistory * การบันทึกอ้างอิง "ลำดับคอลัมน์" ไม่ใช่ชื่อ จึงไม่ต้องแก้โค้ดเมื่อเปลี่ยนชื่อหัวคอลัมน์ * (เช่น Keyoffice -> Product Key) เพียงแต่ลำดับคอลัมน์ในชีตต้องไม่ถูกสลับระหว่างที่เปิดหน้าไว้ */ // ===== ตั้งค่า ===== const CONFIG = { REQUIRE_PASSWORD: false, // false = ไม่ต้องใส่รหัสผ่าน / true = ต้องกรอกรหัสในหน้าเว็บ PASSWORD: 'เปลี่ยนรหัสผ่านตรงนี้', // ใช้เมื่อ REQUIRE_PASSWORD เป็น true SHEET_DATA: 'Asset01', SHEET_HISTORY: 'History', ASSET_HEADER: 'IT Asset No', // หัวคอลัมน์รหัส Asset (ใช้ตรวจแถวก่อนบันทึก) TZ: 'Asia/Bangkok', HISTORY_LIMIT: 300 // จำนวนประวัติสูงสุดที่ส่งกลับต่อเครื่อง }; // ===== จุดเข้า ===== function doGet() { // ถ้ามีไฟล์ HTML ชื่อ "index" ในโปรเจกต์นี้ จะแสดงหน้าเว็บให้เลย try { return HtmlService.createHtmlOutputFromFile('index') .setTitle('IT Asset SNNP') .addMetaTag('viewport', 'width=device-width, initial-scale=1'); } catch (e) { return ContentService.createTextOutput('IT Asset API is running') .setMimeType(ContentService.MimeType.TEXT); } } function doPost(e) { let out; try { const req = JSON.parse((e && e.postData && e.postData.contents) || '{}'); checkKey_(req.key); const fns = { getData: getData_, saveRow: saveRow_, getHistory: getHistory_ }; const f = fns[req.fn]; if (!f) throw new Error('ไม่รู้จักคำสั่ง: ' + req.fn); out = { ok: true, data: f.apply(null, (req.args || []).concat([req.editor || ''])) }; } catch (err) { out = { ok: false, error: String(err && err.message || err) }; } return ContentService.createTextOutput(JSON.stringify(out)) .setMimeType(ContentService.MimeType.JSON); } // ===== ฟังก์ชันสำหรับ google.script.run (เปิดผ่าน Apps Script) ===== function gs_getData(key, who) { checkKey_(key); return getData_(who); } function gs_saveRow(key, who, n, asset, ch) { checkKey_(key); return saveRow_(n, asset, ch, who); } function gs_getHistory(key, who, asset) { checkKey_(key); return getHistory_(asset, who); } // ===== ตรวจรหัสผ่าน ===== function checkKey_(key) { if (!CONFIG.REQUIRE_PASSWORD) return; if (String(key || '') !== String(CONFIG.PASSWORD)) throw new Error('unauthorized'); } // ===== อ่านข้อมูลทั้งหมด ===== // คืน { headers:[...], rows:[{n:เลขแถวในชีต, v:[...ค่าทุกคอลัมน์เป็นข้อความ]}] } function getData_() { const sh = sheet_(CONFIG.SHEET_DATA); const lastRow = sh.getLastRow(), lastCol = sh.getLastColumn(); if (lastRow < 1 || lastCol < 1) return { headers: [], rows: [] }; const values = sh.getRange(1, 1, lastRow, lastCol).getValues(); const headers = values[0].map(h => String(h == null ? '' : h).trim()); const rows = []; for (let r = 1; r < values.length; r++) { const v = values[r].map(cellToText_); if (v.every(x => x === '')) continue; // ข้ามแถวว่างทั้งแถว rows.push({ n: r + 1, v: v }); // n = เลขแถวจริงในชีต (เริ่มที่ 2) } return { headers: headers, rows: rows }; } // ===== บันทึกการแก้ไข ===== // n = เลขแถว, asset = รหัส Asset ของแถวนั้น, ch = { "ลำดับคอลัมน์": "ค่าใหม่", ... } function saveRow_(n, asset, ch, who) { n = Number(n); if (!n || n < 2) throw new Error('เลขแถวไม่ถูกต้อง'); if (!ch || typeof ch !== 'object') throw new Error('ไม่มีข้อมูลที่จะบันทึก'); const lock = LockService.getScriptLock(); lock.waitLock(20000); try { const sh = sheet_(CONFIG.SHEET_DATA); const lastCol = sh.getLastColumn(); const headers = sh.getRange(1, 1, 1, lastCol).getValues()[0].map(h => String(h == null ? '' : h).trim()); const assetCol = headers.indexOf(CONFIG.ASSET_HEADER); if (assetCol < 0) throw new Error('ไม่พบคอลัมน์ ' + CONFIG.ASSET_HEADER); if (n > sh.getLastRow()) throw new Error('ไม่พบแถวที่ ' + n + ' (ข้อมูลอาจถูกแก้ไขแล้ว กด Refresh)'); // กันบันทึกผิดแถว: รหัส Asset ในชีตต้องตรงกับที่หน้าเว็บเปิดอยู่ const rowVals = sh.getRange(n, 1, 1, lastCol).getValues()[0]; const current = cellToText_(rowVals[assetCol]); if (String(asset || '') !== current) { throw new Error('แถวนี้ถูกเปลี่ยนไปแล้ว (' + current + ') กด Refresh แล้วลองใหม่'); } const stamp = Utilities.formatDate(new Date(), CONFIG.TZ, 'yyyy-MM-dd HH:mm:ss'); const log = []; let saved = 0; Object.keys(ch).forEach(k => { const i = Number(k); if (!(i >= 0 && i < lastCol)) return; const oldV = cellToText_(rowVals[i]); const newV = ch[k] == null ? '' : String(ch[k]); if (oldV === newV) return; // ไม่เปลี่ยน ไม่ต้องบันทึก sh.getRange(n, i + 1).setValue(newV); log.push([stamp, current, headers[i], oldV, newV, who || '']); saved++; }); if (log.length) { const hs = sheet_(CONFIG.SHEET_HISTORY); hs.getRange(hs.getLastRow() + 1, 1, log.length, 6).setValues(log); } SpreadsheetApp.flush(); return { saved: saved }; } finally { lock.releaseLock(); } } // ===== ประวัติการแก้ไขของเครื่อง (ใหม่สุดก่อน) ===== // แต่ละรายการ: [วันที่แก้ไข, IT Asset No, ช่อง, ค่าเดิม, ค่าใหม่, ผู้แก้ไข] function getHistory_(asset) { const hs = sheet_(CONFIG.SHEET_HISTORY); const lastRow = hs.getLastRow(); if (lastRow < 2) return []; const rows = hs.getRange(2, 1, lastRow - 1, 6).getValues(); const target = String(asset || ''); const out = []; for (let r = rows.length - 1; r >= 0 && out.length < CONFIG.HISTORY_LIMIT; r--) { if (cellToText_(rows[r][1]) === target) out.push(rows[r].map(cellToText_)); } return out; } // ===== ตัวช่วย ===== function sheet_(name) { const sh = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(name); if (!sh) throw new Error('ไม่พบชีตชื่อ "' + name + '"'); return sh; } function cellToText_(x) { if (x == null) return ''; if (x instanceof Date) return Utilities.formatDate(x, CONFIG.TZ, 'yyyy-MM-dd HH:mm:ss'); if (typeof x === 'number') return Number.isInteger(x) ? String(x) : String(x); return String(x).trim(); }
